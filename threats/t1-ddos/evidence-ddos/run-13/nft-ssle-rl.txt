	set buckets {
		type ipv4_addr
		size 65535

	chain input {
		type filter hook input priority filter - 5; policy accept;
		tcp dport 8000 ct state new ip saddr @limited goto rl
		tcp dport 8000 ct state new ip saddr @limited_all goto rl
	}

	chain rl {
		update @buckets { ip saddr limit rate over 20/second burst 40 packets } counter packets 0 bytes 0 drop
	}
}
